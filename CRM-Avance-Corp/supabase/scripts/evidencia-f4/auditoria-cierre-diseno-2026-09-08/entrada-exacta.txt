FUNCTION crm.crear_contrato_con_cuenta_pdf_v2(jsonb,jsonb,jsonb)
1: 
2: declare
3:   v_actor_id uuid := (select auth.uid());

5:   v_resultado jsonb;

10:   v_huella text;
11:   v_huella_previa text;
12: begin
13:   perform private.inversiones_escritura_bajo_candado(); -- F4: bandera antes de persona/cuenta/PDF
14:   if v_actor_id is null then
15:     raise insufficient_privilege using message = 'Sesión no válida';
16:   end if;
17: 
18:   -- IDEMPOTENCIA DEL ALTA (05/09/2026). El front manda una clave (uuid) por intento de
19:   -- formulario, la MISMA en cada reintento. Si este actor ya registró un alta con esa
20:   -- clave, se devuelve el MISMO contrato en vez de crear otro. Viaja DENTRO de
110:     p_cronograma,
111:     p_cuenta
112:   );

116:     raise exception 'El alta no devolvió un contrato válido'
117:       using errcode = 'P0001';
118:   end;
119:   if v_contrato_id is null then
120:     raise exception 'El alta no devolvió un contrato válido'
121:       using errcode = 'P0001';
122:   end if;
123: 
124:   v_pdf := private.crear_job_contrato_pdf_base(v_contrato_id, v_actor_id);

FUNCTION crm.fusionar_inversionistas_fn(uuid,uuid,text,text)
1: 
2: declare
3:   v_uid uuid;
4:   v_p crm.inversionistas%rowtype; v_c crm.inversionistas%rowtype; v_row crm.inversionistas%rowtype;

12: begin
13:   if not private.resolver_en_puertas_bajo_candado() then
14:     raise exception 'Identidad unificada apagada' using errcode = 'P0409';
15:   end if;
16:   if not private.es_gerencia_crm_activa() then
17:     raise exception 'Solo Gerencia fusiona identidades' using errcode = '42501';
18:   end if;

21:     raise exception 'Indica dos identidades distintas' using errcode = '22023';

24:     raise exception 'Falta la huella de la previsualización (p_hash)' using errcode = '22023';

27:   -- 1. jerarquía compartida [E3-2] y Gerencia REVALIDADA bajo ella -> 2. documentos vigentes de ambas (sin lock, ordenados)

33:   perform private.motivo_sin_documento(p_motivo, (select pg_catalog.array_agg(d.documento_normalizado) from crm.inversionista_identificadores d where d.inversionista_id in (p_perdedora, p_canonica)));
34:   -- 3. identidades FOR UPDATE por id ascendente (P, C y las predecesoras de P: el aplanado no espera después de las reservas [E3-12])
35:   for v_row in select * from crm.inversionistas where id in (p_perdedora, p_canonica) order by id for update loop

49:   v_leads := coalesce((select pg_catalog.array_agg(x order by x) from private.leads_de_identidades(array[p_perdedora, p_canonica]) x), '{}');

54:    where ce.inversionista_id in (p_perdedora, p_canonica) or ce.lead_id = any(v_leads)
55:    order by ce.id for update;

57:   perform 1 from crm.inversion_titulares t
58:    where t.inversionista_id in (p_perdedora, p_canonica)
59:       or t.inversion_id in (select i.id from crm.inversiones i where i.inversionista_id in (p_perdedora, p_canonica))
60:    order by t.id for update;
155:   update crm.cierres_externos ce set inversionista_id = p_canonica
156:    where ce.inversionista_id = p_perdedora
157:       or (v_lead.id is not null and v_lead.inversionista_id = p_perdedora and ce.lead_id = v_lead.id and ce.inversionista_id is null);

173:     end if;
174:   end loop;
175:   update crm.conversion_reservas set inversionista_id = p_canonica where inversionista_id = p_perdedora;
176:   get diagnostics v_n_res = row_count;
177:   v_impacto := pg_catalog.jsonb_build_object('lead_id', v_lead.id, 'lead_reapuntado', v_lead.id is not null and v_lead.inversionista_id = p_perdedora,

FUNCTION private.proteger_titulares_documentales()
1: 
2: declare
3:   v_anterior uuid := case when tg_op in ('UPDATE', 'DELETE') then old.contrato_id end;
4:   v_nuevo uuid := case when tg_op in ('INSERT', 'UPDATE') then new.contrato_id end;
5: begin
6:   if tg_op = 'DELETE'
7:      and current_setting('crm.contrato_pdf_eliminacion_autorizada', true)
8:            = old.contrato_id::text then
9:     return old;
10:   end if;
11:   if tg_op = 'UPDATE' and new is not distinct from old then return new; end if;
12: 
13:   perform private.bloquear_contratos_hijo_documental(v_anterior, v_nuevo);
14:   if (
15:     v_anterior is not null
16:     and (
17:       private.contrato_en_eliminacion(v_anterior)
18:       or private.contrato_documental_congelado(v_anterior)
19:     )
20:     and not private.mutacion_documental_autorizada(v_anterior)
21:   ) or (
22:     v_nuevo is not null
23:     and (
24:       private.contrato_en_eliminacion(v_nuevo)
25:       or private.contrato_documental_congelado(v_nuevo)
26:     )
27:     and not private.mutacion_documental_autorizada(v_nuevo)
28:   ) then
29:     raise exception 'Los titulares están congelados por su PDF legal'
30:       using errcode = '55000';
31:   end if;
32: 
33:   if tg_op = 'DELETE' then return old; end if;
34:   return new;
35: end;
