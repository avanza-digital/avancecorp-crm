908: // El SDK interpreta código estático; la lógica de negocio corre en nodos Code.
909: function literalSDK(valor) {
910:   if (typeof valor === 'string' && valor.startsWith('={{')) return 'expr(' + JSON.stringify(valor.slice(1)) + ')'
911:   if (Array.isArray(valor)) return '[' + valor.map(literalSDK).join(',') + ']'
912:   if (valor && typeof valor === 'object') return '{' + Object.entries(valor).map(([clave, v]) => JSON.stringify(clave) + ':' + literalSDK(v)).join(',') + '}'
913:   return JSON.stringify(valor)
914: }

916: export function codigoSDK(nodos, id = WORKFLOW_ID, nombre = NOMBRE, conexiones = CONEXIONES, grupos = GRUPOS) {
917:   const handles = new Map(nodos.map((nodo, i) => [nodo.name, 'n' + i]))
918:   if (handles.size !== nodos.length) throw new Error('Nombres de nodos repetidos')
919:   const nodoPorNombre = new Map(nodos.map(nodo => [nodo.name, nodo]))
920:   const handle = nombreNodo => {
921:     if (!handles.has(nombreNodo)) throw new Error('Nodo desconocido: ' + nombreNodo)
922:     return handles.get(nombreNodo)
923:   }
924:   const definiciones = nodos.map((nodo, i) => {
925:     const { type, typeVersion, output, ...config } = nodo

156:   -- F2.b [D-4]: solo el importador (service_role, sin sesión de usuario), la misma regla que registrar_reingreso_lead_fn.
157:   if (select auth.uid()) is not null then
158:     raise exception 'Solo el importador (service_role) usa esta puerta' using errcode = '42501';
159:   end if;
160:   -- Defensa en profundidad (auditor N5): esta puerta nunca corre como operación privilegiada, venga como venga la sesión.
161:   perform pg_catalog.set_config('crm.op_privilegiada', 'off', true);
162:   if p_fila is null or pg_catalog.jsonb_typeof(p_fila) <> 'object' then
163:     raise exception 'Fila invalida' using errcode = '22023';
164:   end if;
165:   -- Los casts van DESPUÉS de la autorización (Codex v3 #6): un payload malformado no cambia el 42501 por un 22xxx.
166:   v_campania := private.normalizar_campania_origen(p_fila->'campania_origen', v_origen);
167:   v_fecha    := nullif(pg_catalog.btrim(p_fila->>'fecha_nacimiento'), '')::date;
168:   v_monto    := nullif(pg_catalog.btrim(p_fila->>'monto_estimado'), '')::numeric;
169:   v_cons_en  := nullif(pg_catalog.btrim(p_fila->>'consentimiento_en'), '')::timestamptz;
170:   v_vendedor := nullif(pg_catalog.btrim(p_fila->>'vendedor_id'), '')::uuid;
171:   -- La puerta NO valida el formato: lo hace la fila al nacer (trigger 00: teléfono y DNI → 22023 con el mismo texto de
172:   -- hoy; NOT NULL y CHECKs de la tabla), con la MISMA prioridad que el INSERT directo —el veto de la persona (000) va
173:   -- antes que el formato— (Codex v4 #9). El teléfono viaja crudo, como lo manda el edge; lo normaliza el trigger de la tabla.
174: 
175:   -- Orden TOTAL de candados (b1/D-13, el de crm.crear_lead_si_disponible): documento → persona → contactos → fila.
176:   -- El INSERT directo de hoy YA los toma en ese orden (triggers BEFORE 000 → 00, antes de tocar tupla e índice). Aquí
177:   -- se toman explícitos SOLO los de identidad (documento → persona; apagada la bandera no toman nada), antes del
178:   -- sub-bloque para que sobrevivan a su rollback y el reingreso corra con la persona retenida. Los de CONTACTO los toma
179:   -- el trigger 00 del INSERT, DESPUÉS del veto de la persona (000), exactamente como el INSERT directo: una persona
180:   -- vetada recibe su «no insistir» sin esperar por un teléfono retenido (Codex v3 #3). D-4 no cambia el orden: lo que
181:   -- aporta es un veredicto único sin parsear SQLSTATE y el reingreso en la misma transacción.
182:   perform private.identidad_bloquear_documento('DNI', v_dni);
183:   perform private.identidad_bloquear_persona('DNI', v_dni);

185:   -- El CONTRATO del importador es el de la fila al nacer sin sesión humana (trigger de disponibilidad: «un escritor
186:   -- interno se serializa pero conserva su contrato especializado»): NO se consulta la disponibilidad comercial
187:   -- (enfriamiento, ficha de cliente, «no insistir» de un lead viejo) —el cliente que vuelve por la hoja es el mejor
188:   -- lead posible—; sí mandan el índice único de teléfono VIVO (duplicado) y, con la identidad encendida, los triggers
189:   -- de nacimiento: 000 (la persona vetada → P0429) y zz (la persona ya tiene lead o está en conversión → P0481 por
190:   -- identidad). Por eso la puerta hace el MISMO INSERT (payload exacto del edge: sin sesión, cola global, alta_manual
191:   -- por defecto) y convierte esos tres resultados en un veredicto en vez de una excepción.
192:   begin
193:     insert into crm.leads (
194:       id, nombre_completo, telefono, telefono_alternativo, telefono_alternativo_crudo, correo, dni, genero,
195:       fecha_nacimiento, distrito, origen, campania_origen, etapa, monto_estimado, moneda, categoria_interes, nota, no_contactar,
196:       consentimiento_en, consentimiento_fuente, vendedor_id, asignado_supervisor_id, activo, creado_por
197:     ) values (
198:       v_id, v_nombre, v_telefono, v_alt, v_alt_crudo, v_correo, v_dni, v_genero,
199:       v_fecha, v_distrito, v_origen, v_campania, 'nuevo', v_monto, v_moneda, v_categoria, v_nota, false,
200:       v_cons_en, v_cons_fuente, v_vendedor, null, true, null
201:     );
202:     return pg_catalog.jsonb_build_object('resultado', 'importado', 'lead_id', v_id, 'veredicto', null, 'reingreso', null);
203:   exception
204:     when unique_violation then
205:       -- uq_leads_telefono_vivo (o uq_leads_dni_vivo): ya hay un lead VIVO con ese contacto → DUPLICADO (hoy: 23505).
206:       -- Se devuelve el NOMBRE del índice, no el DETAIL (que lleva el teléfono o el DNI en claro; auditor N1).
207:       get stacked diagnostics v_constraint = constraint_name;
208:       v_veredicto := pg_catalog.jsonb_build_object('estado', 'duplicado', 'indice', coalesce(v_constraint, 'desconocido'));
209:       v_resultado := 'duplicado';
210:     when sqlstate 'P0481' then
211:       -- «Contacto no disponible» con el veredicto en DETAIL (hoy lo parsea el edge): ya_es_cliente por identidad
212:       -- (con lead_id) o en conversión → ya_cliente; cualquier otro → rechazado.
213:       get stacked diagnostics v_detalle = pg_exception_detail;
214:       begin
215:         v_veredicto := v_detalle::jsonb;
216:       exception when others then
217:         v_veredicto := pg_catalog.jsonb_build_object('estado', 'no_disponible', 'detalle', pg_catalog.left(coalesce(v_detalle, ''), 200));
218:       end;
219:       v_resultado := case when v_veredicto->>'estado' = 'ya_es_cliente' and v_veredicto->>'via' = 'identidad' then 'ya_cliente' else 'rechazado' end;
220:     when sqlstate 'P0429' then
221:       -- La persona tiene «No insistir» (trigger 000, identidad encendida) → RECHAZADO (hoy: P0429).
222:       get stacked diagnostics v_detalle = pg_exception_detail;
223:       begin
224:         v_veredicto := coalesce(nullif(v_detalle, '')::jsonb, pg_catalog.jsonb_build_object('estado', 'no_contactar'));
225:       exception when others then
226:         v_veredicto := pg_catalog.jsonb_build_object('estado', 'no_contactar');
227:       end;
228:       v_resultado := 'rechazado';
229:   end;
230: 
231:   v_lead := nullif(v_veredicto->>'lead_id', '')::uuid;
232:   -- F2.b [D-20] (auditor A1): `private.leads_de_personas` devuelve enlace ∪ puente ∪ sueltos, así que `v_lead`
233:   -- también trae leads CONVERTIDOS y descartados. Anotar el reingreso en uno cerrado es escribir donde nadie mira.
234:   -- La bifurcación va por el lead VIVO: solo ése es un trabajo que alguien tiene en su bandeja.
235:   -- Y no basta con mirar el lead que trae el veredicto: si ése está cerrado pero la persona conserva OTRO vivo, crear
236:   -- una tarea duplicaría el trabajo (Codex). Se busca cualquier lead vivo suyo, prefiriendo el del veredicto.
237:   select l.id into v_lead_vivo
238:     from crm.leads l
239:    where l.activo
240:      and l.etapa not in ('convertido', 'descartado')
241:      and (l.id = v_lead
242:           or (nullif(pg_catalog.btrim(coalesce(v_dni, '')), '') is not null and l.dni = v_dni)
243:           or (l.inversionista_id is not null
244:               and l.inversionista_id = private.inversionista_por_documento('DNI', v_dni)))
245:    order by (l.id = v_lead) desc, l.creado_en desc
246:    limit 1;
247:   if v_resultado = 'ya_cliente' and v_lead_vivo is not null then
248:     -- El reingreso en la MISMA transacción (hoy el edge lo pedía aparte tras leer el error). Si fallara de forma
249:     -- DEFINITIVA, la fila sigue siendo «ya cliente» y el edge lo dice en la hoja, como hoy.
250:     v_datos := pg_catalog.jsonb_build_object(
251:       'fila', p_fila->'fila', 'nombre', v_nombre, 'telefono', v_telefono, 'telefono_alternativo', v_alt,
252:       'correo', v_correo, 'capital', v_monto, 'moneda', v_moneda, 'canal', v_origen, 'distrito', v_distrito,
253:       'interes', v_categoria, 'nota', v_nota);
254:     -- Ausente no añade una clave nula: se conserva la igualdad de solicitudes antiguas.
255:     if v_campania is not null then
256:       v_datos := v_datos || pg_catalog.jsonb_build_object('campania_origen', v_campania);
257:     end if;
258:     -- IDEMPOTENCIA (Codex v4 #7, v5 #3): si la hoja reenvía la MISMA fila (perdió la confirmación HTTP o el lote falló
259:     -- después), el reingreso ya anotado vale: no se escribe una segunda nota. Misma persona, mismo origen y los mismos
260:     -- DATOS —sin el número de fila, que cambia si alguien inserta una fila encima— en los últimos 7 días (una caída larga
261:     -- del CRM tampoco la burla). Es lo que registrar_reingreso_lead_fn guarda en metadata.datos.
262:     select a.id into v_prev
263:       from crm.actividades a
264:      where a.lead_id = v_lead_vivo
265:        and a.metadata->>'evento' = 'reingreso'
266:        and a.metadata->>'origen' = 'hoja'
267:        and (a.metadata->'datos') - 'fila' = v_datos - 'fila'
268:        and a.creado_en > pg_catalog.now() - interval '7 days'
269:      order by a.creado_en desc
270:      limit 1;
271:     if v_prev is not null then
272:       v_reingreso := pg_catalog.jsonb_build_object('ok', true, 'actividad_id', v_prev, 'repetido', true);
273:     else
274:       begin
275:         v_reingreso := crm.registrar_reingreso_lead_fn(v_lead_vivo, 'hoja', v_datos);
276:       exception when others then
277:         -- Solo los errores DE NEGOCIO quedan anotados como definitivos en la respuesta (datos 22xxx, integridad 23xxx y
278:         -- los P0xxx que levantan nuestras funciones: lead inexistente, origen inválido, identidad apagada). Cualquier
279:         -- otro —transacción, recursos, candado, cancelación, E/S, límites (54), internos, configuración— sube entero: el
280:         -- edge lo trata como temporal y la hoja reintenta la fila (auditor M4; Codex v4 #8 y v5 #4).
281:         if pg_catalog.left(sqlstate, 2) not in ('22', '23', 'P0') then
282:           raise;
283:         end if;
284:         v_reingreso := pg_catalog.jsonb_build_object('ok', false, 'error', sqlstate || ': ' || pg_catalog.left(sqlerrm, 120));
285:       end;
286:     end if;
287:   elsif v_resultado = 'ya_cliente' then
288:     -- F2.b [D-20]: es cliente y NO conserva lead vivo (el caso de 407 de los 424 clientes activos en producción al
289:     -- 06/09). Sin esta rama la solicitud se perdía: la hoja la daba por cerrada y en el CRM no quedaba nada. Se deja
290:     -- la huella donde el analista mira: nota en la ficha del cliente y tarea de llamada para el responsable de su
291:     -- cartera. Nunca las dos vías a la vez: si hubiera lead vivo, la rama de arriba ya anotó el reingreso.
292:     v_datos := pg_catalog.jsonb_build_object(
293:       'fila', p_fila->'fila', 'nombre', v_nombre, 'telefono', v_telefono, 'telefono_alternativo', v_alt,
294:       'correo', v_correo, 'capital', v_monto, 'moneda', v_moneda, 'canal', v_origen, 'distrito', v_distrito,
295:       'interes', v_categoria, 'nota', v_nota);
296:     -- Ausente no añade una clave nula: se conserva la igualdad de solicitudes antiguas.
297:     if v_campania is not null then
298:       v_datos := v_datos || pg_catalog.jsonb_build_object('campania_origen', v_campania);
299:     end if;
300:     -- La persona ya está reconocida (el veredicto lo dice), así que su ficha de cliente sale de la identidad. El
301:     -- respaldo por documento exacto cubre a un cliente cuya identidad todavía no lleva el perfil enlazado.
302:     select i.perfil_id into v_perfil
303:       from crm.inversionistas i
304:      where i.id = private.inversionista_por_documento('DNI', v_dni) and i.perfil_id is not null;
305:     if v_perfil is null and nullif(pg_catalog.btrim(coalesce(v_dni, '')), '') is not null then
306:       select p.id into v_perfil
307:         from public.perfiles p
308:        where p.rol = 'cliente' and p.activo
309:          and nullif(pg_catalog.btrim(coalesce(p.dni, '')), '') is not null
310:          and (coalesce(nullif(pg_catalog.btrim(p.tipo_documento), ''), 'DNI') || ':' ||
311:               pg_catalog.upper(pg_catalog.regexp_replace(p.dni, '[^A-Za-z0-9]', '', 'g')))
312:              = ('DNI:' || pg_catalog.upper(pg_catalog.regexp_replace(coalesce(v_dni, ''), '[^A-Za-z0-9]', '', 'g')))
313:        limit 1;
314:     end if;
315:     if v_perfil is not null then
316:       begin
317:         v_solicitud := private.registrar_solicitud_cliente_fn(v_perfil, 'hoja', v_datos);
318:       exception when others then
319:         -- Mismo criterio que el reingreso: solo los errores DE NEGOCIO se anotan como definitivos; cualquier otro
320:         -- sube entero para que la hoja reintente la fila.
321:         if pg_catalog.left(sqlstate, 2) not in ('22', '23', 'P0') then
322:           raise;
323:         end if;
324:         v_solicitud := pg_catalog.jsonb_build_object('ok', false, 'error', sqlstate || ': ' || pg_catalog.left(sqlerrm, 120));
325:       end;
326:     else
327:       v_solicitud := pg_catalog.jsonb_build_object('ok', false, 'error', 'no se pudo ubicar la ficha del cliente');
328:     end if;
329:   end if;
330: 
331:   return pg_catalog.jsonb_build_object('resultado', v_resultado, 'lead_id', v_lead, 'veredicto', v_veredicto, 'reingreso', v_reingreso)
332:       || case when v_solicitud is null then '{}'::jsonb else pg_catalog.jsonb_build_object('solicitud', v_solicitud) end;
333: end;

336: CREATE OR REPLACE FUNCTION crm.registrar_reingreso_lead_fn(p_lead_id uuid, p_origen text, p_datos jsonb DEFAULT '{}'::jsonb)
337:  RETURNS jsonb
338:  LANGUAGE plpgsql
339:  SECURITY DEFINER
340:  SET search_path TO ''
341: AS $function$
342: declare
343:   v_id uuid;
344:   v_datos jsonb;
345: begin
346:   if (select auth.uid()) is not null then
347:     raise exception 'Solo el importador (service_role) registra reingresos' using errcode = '42501';
348:   end if;
349:   -- Paridad apagada: superficie inerte mientras la identidad no esté activa (Codex E1).
350:   if not private.resolver_en_puertas_bajo_candado() then
351:     raise exception 'Identidad unificada apagada: el reingreso no se registra' using errcode = 'P0409';
352:   end if;
353:   if p_lead_id is null or not exists (select 1 from crm.leads l where l.id = p_lead_id) then
354:     raise exception 'Lead inexistente' using errcode = 'P0002';
355:   end if;
356:   if p_origen is null or pg_catalog.length(pg_catalog.btrim(p_origen)) not between 1 and 40 then
357:     raise exception 'Origen inválido' using errcode = '22023';
358:   end if;
359:   -- Sin documento en claro en la actividad: solo datos comerciales del formulario.
360:   v_datos := (
361:     select coalesce(pg_catalog.jsonb_object_agg(e.key, e.value), '{}'::jsonb)
362:     from pg_catalog.jsonb_each(coalesce(p_datos, '{}'::jsonb)) e
363:     where e.key in ('nombre','telefono','telefono_alternativo','correo','capital','moneda','canal','distrito','interes','nota','fila','campania_origen')
364:   );
365:   insert into crm.actividades (lead_id, tipo, detalle, metadata, creado_por)
366:   values (
367:     p_lead_id, 'nota',
368:     'Reingreso por ' || pg_catalog.btrim(p_origen) || ': la persona volvió a dejar sus datos'
369:       || coalesce(' · campaña: ' || nullif(v_datos->>'campania_origen', ''), ''),
370:     pg_catalog.jsonb_build_object('evento', 'reingreso', 'origen', pg_catalog.btrim(p_origen), 'datos', v_datos),
371:     null)
372:   returning id into v_id;
373:   return pg_catalog.jsonb_build_object('ok', true, 'actividad_id', v_id);
374: end;

377: CREATE OR REPLACE FUNCTION private.registrar_solicitud_cliente_fn(p_perfil uuid, p_origen text, p_datos jsonb)
378:  RETURNS jsonb
379:  LANGUAGE plpgsql
380:  SECURITY DEFINER
381:  SET search_path TO ''
382:  SET lock_timeout TO '5s'
383: AS $function$
384: declare
385:   v_cliente   public.perfiles%rowtype;
386:   v_huella    text;
387:   v_detalle   text;
388:   v_prev_act  uuid;
389:   v_act       uuid;
390:   v_tarea     uuid;
391:   v_sin_asesor boolean := false;
392:   v_motivo    text;
393:   v_dest      uuid;      -- a nombre de quién queda la nota (la tabla exige responsable)
394: begin
395:   -- F2.b [D-20]. Solo el importador, la misma regla que registrar_reingreso_lead_fn: nunca una sesión humana.
396:   if (select auth.uid()) is not null then
397:     raise exception 'Solo el importador (service_role) registra solicitudes de cliente' using errcode = '42501';
398:   end if;
399:   if not private.resolver_en_puertas_bajo_candado() then
400:     raise exception 'Identidad unificada apagada: la solicitud no se registra' using errcode = 'P0409';
401:   end if;
402:   if p_origen is null or p_origen not in ('hoja', 'landing', 'web') then
403:     raise exception 'Origen inválido' using errcode = '22023';
404:   end if;
405:   select * into v_cliente from public.perfiles p where p.id = p_perfil and p.rol = 'cliente';
406:   if not found then
407:     raise exception 'Cliente inexistente' using errcode = 'P0002';
408:   end if;
409:   if not v_cliente.activo then
410:     raise exception 'Cliente inactivo: la solicitud no se registra' using errcode = 'P0409';
411:   end if;
412:   -- Auditor D-20 #A2: los dos triggers de veto se EXIMEN cuando no hay sesión (auth.uid() null), que es justo el caso
413:   -- del importador. La ley 29571 no puede depender de un razonamiento sobre el orden de los triggers: se comprueba aquí.
414:   if private.persona_vetada_perfil(p_perfil) then
415:     return pg_catalog.jsonb_build_object('ok', false, 'vetado', true, 'sin_tarea', true,
416:       'error', 'La persona tiene la restricción «No insistir»: no se registra seguimiento');
417:   end if;
418:   -- Auditor #M6: la idempotencia se sostiene sola. Hoy el importador ya serializa por persona, pero el ayudante no
419:   -- puede depender de quién lo llame.
420:   perform pg_catalog.pg_advisory_xact_lock(pg_catalog.hashtext('solicitud_cliente:' || p_perfil::text));
421: 
422:   -- Huella de la solicitud SIN el número de fila (que cambia si alguien inserta una fila encima), igual que el
423:   -- reingreso: si la hoja reenvía la misma fila porque perdió la confirmación, no se duplica ni la nota ni la tarea.
424:   v_huella := pg_catalog.md5(coalesce((p_datos - 'fila')::text, ''));
425:   v_detalle := 'Pidió información por la web'
426:     -- Antes de la nota larga; la huella sigue al final y conserva la campaña.
427:     || coalesce(' · campaña: ' || nullif(p_datos->>'campania_origen', ''), '')
428:     || coalesce(' · interés: ' || nullif(pg_catalog.btrim(coalesce(p_datos->>'interes', '')), ''), '')
429:     || coalesce(' · capital: ' || nullif(pg_catalog.btrim(coalesce(p_datos->>'capital', '')), ''), '')
430:     || coalesce(' ' || nullif(pg_catalog.btrim(coalesce(p_datos->>'moneda', '')), ''), '')
431:     || coalesce(' · distrito: ' || nullif(pg_catalog.btrim(coalesce(p_datos->>'distrito', '')), ''), '')
432:     || coalesce(' · canal: ' || nullif(pg_catalog.btrim(coalesce(p_datos->>'canal', '')), ''), '')
433:     || coalesce(' · teléfono: ' || nullif(pg_catalog.btrim(coalesce(p_datos->>'telefono', '')), ''), '')
434:     || coalesce(' · nota: ' || nullif(pg_catalog.btrim(coalesce(p_datos->>'nota', '')), ''), '');
435:   -- `detalle` y `nota` topan en 2000 caracteres, pero la nota del formulario no tiene límite: se recorta el cuerpo y
436:   -- la huella se añade DESPUÉS, porque recortar por el final la destruiría y con ella la idempotencia (auditor #M1).
437:   v_detalle := pg_catalog.left(v_detalle, 1900) || ' · Ref: ' || v_huella;
438: 
439:   select a.id into v_prev_act
440:     from crm.actividades_cliente a
441:    where a.cliente_id = p_perfil
442:      and a.tipo = 'nota'
443:      -- La comparación va contra la COLA, no contra el interior (Codex): el sufijo « · Ref: <huella>» lo añadimos
444:      -- nosotros SIEMPRE al final, mientras que la nota del visitante queda antes y recortada. Con `strpos`, un texto
445:      -- del formulario que contuviera «Ref: <huella>» podía hacer pasar por repetida una solicitud jamás registrada.
446:      and pg_catalog.right(coalesce(a.detalle, ''), pg_catalog.length(' · Ref: ' || v_huella)) = ' · Ref: ' || v_huella
447:      and a.creado_por is null   -- auditor #M2: solo cuenta la nota que escribió el importador, no una copiada a mano
448:      and a.creado_en > pg_catalog.now() - interval '7 days'
449:    order by a.creado_en desc
450:    limit 1;
451:   if v_prev_act is not null then
452:     return pg_catalog.jsonb_build_object('ok', true, 'actividad_id', v_prev_act, 'repetido', true);
453:   end if;
454: 
455:   -- `actividades_cliente` exige responsable en toda nota (CHECK) y que ese responsable esté en el equipo (FK). El
456:   -- analista de la cartera es el destino natural; si el perfil no lo tiene, se usa el responsable VIVO de su identidad
457:   -- (423 de los 424 clientes activos lo tienen). Sin ninguno de los dos no hay a quién atribuirla: se dice y ya.
458:   v_dest := v_cliente.asesor_perfil_id;
459:   if v_dest is null or not exists (select 1 from crm.equipo e where e.perfil_id = v_dest and e.activo and e.rol_crm in ('vendedor','supervisor')) then
460:     select r.responsable_id into v_dest
461:       from crm.inversionista_responsables r
462:       join crm.inversionistas i on i.id = r.inversionista_id
463:      where i.perfil_id = p_perfil and r.hasta is null
464:        and exists (select 1 from crm.equipo e where e.perfil_id = r.responsable_id and e.activo and e.rol_crm in ('vendedor','supervisor'))
465:      order by r.desde desc
466:      limit 1;
