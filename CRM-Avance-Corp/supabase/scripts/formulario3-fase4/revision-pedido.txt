35:   let stdout;
36:   try {
37:     ({ stdout } = await ejecutarCLI('supabase', [
38:       'db', 'query', '--linked', '--project-ref', proyecto, '--output', 'json', sql,
39:     ], { timeout: 90_000, maxBuffer: 256 * 1024, encoding: 'utf8' }));
40:   } catch {
